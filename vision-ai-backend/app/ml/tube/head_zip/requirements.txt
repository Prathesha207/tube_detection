# Tube head/tail/tubing detection pipeline - requirements
#
# Install PyTorch FIRST with the CUDA build matching your GPU driver, then
# everything else. On your RTX 4050, this is the correct index:
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements.txt
#
# Installing in the other order can silently downgrade torch back to the
# CPU-only build, since ultralytics pulls in a CPU torch as a dependency.

numpy>=1.24               # array math throughout
scipy>=1.10                # used by scikit-image internally
scikit-image>=0.21         # skeletonize() - tube axis / width from masks
pytest>=7.4                # only for running test_measure.py
